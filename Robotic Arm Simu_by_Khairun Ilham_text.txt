/* =====================================================
   ROBOT PARAMETERS
===================================================== */

let L1 = 120;
let L2 = 100;
let L3 = 80;


/* =====================================================
   HTML ELEMENTS
===================================================== */

const canvas =
    document.getElementById("robotCanvas");

const ctx =
    canvas.getContext("2d");


const joint1 =
    document.getElementById("joint1");

const joint2 =
    document.getElementById("joint2");

const joint3 =
    document.getElementById("joint3");


const link1 =
    document.getElementById("link1");

const link2 =
    document.getElementById("link2");

const link3 =
    document.getElementById("link3");


const j1Value =
    document.getElementById("j1Value");

const j2Value =
    document.getElementById("j2Value");

const j3Value =
    document.getElementById("j3Value");


const xInput =
    document.getElementById("xInput");

const yInput =
    document.getElementById("yInput");

const zInput =
    document.getElementById("zInput");


const workspaceStatus =
    document.getElementById("workspaceStatus");


/* =====================================================
   DEGREE / RADIAN
===================================================== */

function degToRad(deg){

    return deg * Math.PI / 180;

}

function radToDeg(rad){

    return rad * 180 / Math.PI;

}


/* =====================================================
   FORWARD KINEMATICS
===================================================== */

function calculatePosition(){

    const theta1 =
        degToRad(Number(joint1.value));

    const theta2 =
        degToRad(Number(joint2.value));

    const theta3 =
        degToRad(Number(joint3.value));


    const horizontal =
        L2 * Math.cos(theta2)
        +
        L3 * Math.cos(theta2 + theta3);


    const x =
        horizontal * Math.cos(theta1);


    const y =
        horizontal * Math.sin(theta1);


    const z =
        L1
        +
        L2 * Math.sin(theta2)
        +
        L3 * Math.sin(theta2 + theta3);


    return {
        x:x,
        y:y,
        z:z,

        theta1:theta1,
        theta2:theta2,
        theta3:theta3
    };

}


/* =====================================================
   UPDATE ROBOT PARAMETERS
===================================================== */

function updateLengths(){

    L1 =
        Math.max(
            20,
            Number(link1.value) || 20
        );

    L2 =
        Math.max(
            20,
            Number(link2.value) || 20
        );

    L3 =
        Math.max(
            20,
            Number(link3.value) || 20
        );

}


/* =====================================================
   UPDATE SIMULATOR
===================================================== */

function updateSimulation(){

    updateLengths();


    const data =
        calculatePosition();


    j1Value.textContent =
        joint1.value + "°";

    j2Value.textContent =
        joint2.value + "°";

    j3Value.textContent =
        joint3.value + "°";


    /*
       Tampilkan posisi aktual
       ke input X Y Z
    */

    xInput.value =
        data.x.toFixed(1);

    yInput.value =
        data.y.toFixed(1);

    zInput.value =
        data.z.toFixed(1);


    checkWorkspace();


    drawRobot();

}


/* =====================================================
   WORKSPACE CHECK
===================================================== */

function checkWorkspace(){

    const data =
        calculatePosition();


    const horizontal =
        Math.sqrt(
            data.x * data.x +
            data.y * data.y
        );


    const vertical =
        data.z - L1;


    const distance =
        Math.sqrt(
            horizontal * horizontal +
            vertical * vertical
        );


    const maxReach =
        L2 + L3;


    const minReach =
        Math.abs(L2 - L3);


    if(
        distance <= maxReach + 0.01 &&
        distance >= minReach - 0.01 &&
        data.z >= 0
    ){

        workspaceStatus.className =
            "workspace safe";

        workspaceStatus.textContent =
            "● END EFFECTOR IN WORKSPACE";

    }else{

        workspaceStatus.className =
            "workspace warning";

        workspaceStatus.textContent =
            "● OUTSIDE WORKSPACE";

    }

}


/* =====================================================
   INVERSE KINEMATICS
===================================================== */

function solveInverseKinematics(
    targetX,
    targetY,
    targetZ
){

    /*
       J1
       Base rotation
    */

    const theta1 =
        Math.atan2(
            targetY,
            targetX
        );


    /*
       Horizontal distance
    */

    const r =
        Math.sqrt(
            targetX * targetX +
            targetY * targetY
        );


    /*
       Height relatif terhadap base
    */

    const z =
        targetZ - L1;


    /*
       Distance from shoulder
    */

    const D =
        (
            r * r +
            z * z -
            L2 * L2 -
            L3 * L3
        )
        /
        (
            2 * L2 * L3
        );


    /*
       Cek apakah target
       bisa dicapai
    */

    if(
        D < -1 ||
        D > 1
    ){

        return null;

    }


    /*
       Elbow angle
    */

    const theta3 =
        Math.atan2(
            Math.sqrt(
                Math.max(
                    0,
                    1 - D * D
                )
            ),
            D
        );


    /*
       Shoulder angle
    */

    const theta2 =
        Math.atan2(
            z,
            r
        )
        -
        Math.atan2(
            L3 * Math.sin(theta3),
            L2 +
            L3 * Math.cos(theta3)
        );


    return {

        theta1:
            radToDeg(theta1),

        theta2:
            radToDeg(theta2),

        theta3:
            radToDeg(theta3)

    };

}


/* =====================================================
   APPLY TARGET POSITION
===================================================== */

function applyTargetPosition(){

    updateLengths();


    const targetX =
        Number(xInput.value);

    const targetY =
        Number(yInput.value);

    const targetZ =
        Number(zInput.value);


    if(
        !Number.isFinite(targetX) ||
        !Number.isFinite(targetY) ||
        !Number.isFinite(targetZ)
    ){

        workspaceStatus.className =
            "workspace warning";

        workspaceStatus.textContent =
            "● INVALID TARGET POSITION";

        return;

    }


    const solution =
        solveInverseKinematics(
            targetX,
            targetY,
            targetZ
        );


    if(solution === null){

        workspaceStatus.className =
            "workspace warning";

        workspaceStatus.textContent =
            "● TARGET OUTSIDE ROBOT REACH";

        return;

    }


    /*
       Batasi sesuai slider
    */

    const finalJ1 =
        Math.max(
            -180,
            Math.min(
                180,
                solution.theta1
            )
        );


    const finalJ2 =
        Math.max(
            -90,
            Math.min(
                90,
                solution.theta2
            )
        );


    const finalJ3 =
        Math.max(
            -120,
            Math.min(
                120,
                solution.theta3
            )
        );


    /*
       Update slider
    */

    joint1.value =
        finalJ1;

    joint2.value =
        finalJ2;

    joint3.value =
        finalJ3;


    /*
       Update robot
    */

    updateSimulation();


    /*
       Cek apakah sudut
       benar-benar masih sesuai
       dengan batas robot
    */

    const result =
        calculatePosition();


    const error =
        Math.sqrt(
            Math.pow(
                result.x - targetX,
                2
            )
            +
            Math.pow(
                result.y - targetY,
                2
            )
            +
            Math.pow(
                result.z - targetZ,
                2
            )
        );


    if(error < 1){

        workspaceStatus.className =
            "workspace safe";

        workspaceStatus.textContent =
            "● TARGET POSITION REACHED";

    }else{

        workspaceStatus.className =
            "workspace warning";

        workspaceStatus.textContent =
            "● TARGET LIMITED BY JOINT RANGE";

    }

}


/* =====================================================
   CANVAS RESIZE
===================================================== */

function resizeCanvas(){

    const rect =
        canvas.getBoundingClientRect();


    const ratio =
        window.devicePixelRatio || 1;


    canvas.width =
        rect.width * ratio;

    canvas.height =
        rect.height * ratio;


    ctx.setTransform(
        ratio,
        0,
        0,
        ratio,
        0,
        0
    );


    drawRobot();

}


window.addEventListener(
    "resize",
    resizeCanvas
);


/* =====================================================
   DRAW GRID
===================================================== */

function drawGrid(
    centerX,
    centerY
){

    ctx.save();

    ctx.lineWidth = 1;

    for(
        let i=-600;
        i<=600;
        i+=40
    ){

        ctx.strokeStyle =
            "rgba(100,125,140,.14)";


        ctx.beginPath();

        ctx.moveTo(
            centerX + i,
            centerY + 20
        );

        ctx.lineTo(
            centerX + i * .55,
            centerY + 220
        );

        ctx.stroke();

    }


    for(
        let i=0;
        i<=250;
        i+=30
    ){

        ctx.strokeStyle =
            "rgba(100,125,140,.14)";


        ctx.beginPath();

        ctx.moveTo(
            centerX - 600 + i*.5,
            centerY+i
        );

        ctx.lineTo(
            centerX + 600 - i*.5,
            centerY+i
        );

        ctx.stroke();

    }

    ctx.restore();

}

/* =====================================================
   DRAW LINK
===================================================== */

function drawLink(
    x1,
    y1,
    x2,
    y2,
    width
){

    ctx.save();


    /*
       Shadow
    */

    ctx.strokeStyle=
        "rgba(0,0,0,.55)";

    ctx.lineWidth=
        width+8;

    ctx.lineCap="round";


    ctx.beginPath();

    ctx.moveTo(
        x1+5,
        y1+7
    );

    ctx.lineTo(
        x2+5,
        y2+7
    );

    ctx.stroke();


    /*
       Metal body
    */

    const gradient=
        ctx.createLinearGradient(
            x1,
            y1,
            x2,
            y2
        );


    gradient.addColorStop(
        0,
        "#68757d"
    );

    gradient.addColorStop(
        .45,
        "#a1adb2"
    );

    gradient.addColorStop(
        .55,
        "#606d75"
    );

    gradient.addColorStop(
        1,
        "#343f46"
    );


    ctx.strokeStyle=
        gradient;

    ctx.lineWidth=
        width;

    ctx.beginPath();

    ctx.moveTo(
        x1,
        y1
    );

    ctx.lineTo(
        x2,
        y2
    );

    ctx.stroke();


    /*
       Highlight
    */

    ctx.strokeStyle=
        "rgba(255,255,255,.15)";

    ctx.lineWidth=3;

    ctx.beginPath();

    ctx.moveTo(
        x1,
        y1-2
    );

    ctx.lineTo(
        x2,
        y2-2
    );

    ctx.stroke();


    ctx.restore();

}

/* =====================================================
   MAIN ROBOT DRAW
===================================================== */

function drawRobot(){

    const width =
        canvas.clientWidth;

    const height =
        canvas.clientHeight;


    ctx.clearRect(
        0,
        0,
        width,
        height
    );


    const centerX =
        width/2;

    const baseY =
        height*.72;


    /*
       Adaptive scale

       Robot automatically
       adjusts when link lengths
       become larger.
    */

    const totalLength =
        L1+L2+L3;


    const scale =
        Math.min(
            width/(totalLength*2.4),
            height/(totalLength*2.0)
        );


    drawGrid(
        centerX,
        baseY
    );


    drawAxes(
        centerX,
        baseY-45
    );


    drawBase(
        centerX,
        baseY
    );


    const data =
        calculatePosition();


    const t1 =
        data.theta1;

    const t2 =
        data.theta2;

    const t3 =
        data.theta3;


    /*
       Shoulder
    */

    const shoulderX =
        centerX;

    const shoulderY =
        baseY -
        L1*scale;


    /*
       Link 2
    */

    const horizontal2 =
        L2*
        Math.cos(t2)*
        scale;

    const vertical2 =
        L2*
        Math.sin(t2)*
        scale;


    const elbowX =
        shoulderX
        +
        horizontal2*
        Math.cos(t1)
        -
        horizontal2*
        Math.sin(t1)*.28;


    const elbowY =
        shoulderY -
        vertical2;


    /*
       Link 3
    */

    const totalAngle =
        t2+t3;


    const horizontal3 =
        L3*
        Math.cos(totalAngle)*
        scale;


    const vertical3 =
        L3*
        Math.sin(totalAngle)*
        scale;


    const wristX =
        elbowX
        +
        horizontal3*
        Math.cos(t1)
        -
        horizontal3*
        Math.sin(t1)*.28;


    const wristY =
        elbowY -
        vertical3;


    /*
       Draw links
    */

    drawLink(
        shoulderX,
        shoulderY,
        elbowX,
        elbowY,
        Math.max(12,28*scale)
    );


    drawLink(
        elbowX,
        elbowY,
        wristX,
        wristY,
        Math.max(10,23*scale)
    );


    /*
       Joints
    */

    drawJoint(
        shoulderX,
        shoulderY,
        Math.max(9,18*scale)
    );


    drawJoint(
        elbowX,
        elbowY,
        Math.max(8,15*scale)
    );


    drawJoint(
        wristX,
        wristY,
        Math.max(6,11*scale)
    );


    /*
       Gripper
    */

    drawGripper(
        wristX,
        wristY,
        -totalAngle
    );


    /*
       End effector
    */

    ctx.beginPath();

    ctx.arc(
        wristX,
        wristY,
        4,
        0,
        Math.PI*2
    );

    ctx.fillStyle=
        "#ff9d2e";

    ctx.fill();


    /*
       Label
    */

    ctx.fillStyle=
        "#ffb35b";

    ctx.font=
        "11px Arial";

    ctx.fillText(
        "END EFFECTOR",
        wristX+20,
        wristY-14
    );


    /*
       Robot label
    */

    ctx.fillStyle=
        "rgba(210,220,225,.55)";

    ctx.font=
        "10px Arial";

    ctx.fillText(
        "INDUSTRIAL 3-DOF ROBOT",
        20,
        25
    );


    /*
       Length information
    */

    ctx.fillStyle=
        "rgba(210,220,225,.5)";

    ctx.fillText(
        "L1: "+L1+" mm",
        20,
        45
    );

    ctx.fillText(
        "L2: "+L2+" mm",
        20,
        61
    );

    ctx.fillText(
        "L3: "+L3+" mm",
        20,
        77
    );


    /*
       Scale
    */

    ctx.strokeStyle=
        "rgba(180,195,205,.45)";

    ctx.lineWidth=2;

    const scaleLength =
        100*scale;


    ctx.beginPath();

    ctx.moveTo(
        25,
        height-30
    );

    ctx.lineTo(
        25+scaleLength,
        height-30
    );

    ctx.stroke();


    ctx.fillStyle=
        "rgba(180,195,205,.55)";

    ctx.fillText(
        "100 mm",
        25,
        height-37
    );

}
